cmake_minimum_required(VERSION 3.24)
project(shadps4_cache_runtime_test LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
file(READ "${CMAKE_CURRENT_LIST_DIR}/../../project.json" IFSS_TEST_PROJECT)
string(JSON IFSS_TEST_DEPENDENCIES GET "${IFSS_TEST_PROJECT}" dependencies_root)
get_filename_component(IFSS_TEST_DEPENDENCIES "${IFSS_TEST_DEPENDENCIES}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}/../..")
foreach(test IN ITEMS cache_runtime runtime_memo compile_readiness buffer_access range_set input_port lru_cache)
    set(source "test_${test}.cpp")
    if(test STREQUAL "input_port")
        set(source ../input/test_mouse_input.cpp)
    endif()
    set(target "shadps4_${test}_test")
    add_executable(${target} ${source})
    target_compile_features(${target} PRIVATE cxx_std_23)
    target_include_directories(${target} PRIVATE
        ../../src "${IFSS_TEST_DEPENDENCIES}/xxhash" "${IFSS_TEST_DEPENDENCIES}/ext-boost")
    target_compile_definitions(${target} PRIVATE XXH_INLINE_ALL WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(${target} PRIVATE Threads::Threads)
    add_test(NAME ${test} COMMAND ${target} "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(${test} PROPERTIES TIMEOUT 10)
endforeach()
target_sources(shadps4_cache_runtime_test PRIVATE ../../src/video_core/cache_file.cpp)
set_tests_properties(cache_runtime PROPERTIES TIMEOUT 30)
